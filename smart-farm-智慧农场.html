智慧农场

用光敏电阻自动控制照明灯，用土壤湿度传感器（AO + DO）自动控制水泵浇水，并在 LCD1602 上实时显示光照、土壤湿度与设备状态。

套件：SG-346 顶配豪华版（不带主板） 主控：Arduino UNO 难度：进阶

1接线说明

公共电源：面包板 5V / GND 分配

这个项目需要 5V 的元件不止一个（土壤湿度、光敏、LCD、继电器、水泵共 5 路），而 Arduino 只有 1~2 个 5V 引脚。用面包板把 5V 和 GND 展开成两条“母线”，一个 5V 引脚就能分出很多个口：

面包板接电源连接的元件（并联在这条排上）
红色 + 排Arduino 5V土壤 VCC · 光敏 VCC · LCD VCC · 继电器 DC+ · 泵红线
蓝色/黑色 − 排Arduino GND土壤 GND · 光敏 GND · LCD GND · 继电器 DC- · LED 阴极 · 继电器 NO
面包板的红、蓝长条内部本来就是整条连通的，所以“一根 5V 线进来、很多个 5V 点出去”。 若水泵电流较大，建议水泵改由外接 5V 电源供电（如手机充电器），并把外接电源负极与 Arduino GND 共地，减轻 Arduino 的供电压力。

⑥ NodeMCU (ESP8266) —— WiFi 网络模块（新增）

NodeMCU 负责开 WiFi 热点 + 网页服务器，与 Arduino 通过 串口(UART)通信。Arduino 每 2 秒把传感器数据发给 NodeMCU， 网页上的亮度滑块再由 NodeMCU 发命令给 Arduino 调整灯亮度。

⚠ 重点三项：① Uno 的引脚 12 输出 5V，不能直连 NodeMCU 的 RX（ESP8266 仅 3.3V 耐压），必须用 1kΩ/2kΩ 分压或电平转换模块；NodeMCU 的 TX 是 3.3V 输出，直连 Uno 引脚 4 即可。② NodeMCU 的 USB 只用于烧录程序，烧完拔掉，运行时靠 7V 电池供电。③ NodeMCU 的 USB 串口与它的 TX/RX 是同一路 UART0，串口对话时务必拔掉 USB，否则双方抢串口。

2原理

光敏电阻模块：光照越强，光敏电阻阻值越小，分压后模拟输出 AO 的电压随之变化。Arduino 用 analogRead() 读到 0–1023 的数值，低于阈值就判定为 “天黑”，点亮照明灯；反之关灯。

土壤湿度传感器：探头上的两条电极之间，土壤越湿润导电性越好， 模拟输出 AO 的值越小（干 → 数值高，湿 → 数值低）。 程序先读 AO，再按标定的 DRY_VALUE / WET_VALUE 映射成 0–100% 的湿度百分比，并给出“过干 / 适中 / 过湿”的综合判断。模块的 DO 是一个数字开关量，翻转阈值由模块上的蓝色电位器决定， 可直接得到“干燥 / 湿润”两种状态；两者都打印到串口，方便对照与校准。湿度低于 30% 时接通继电器抽水，同时板载 LED（D13）点亮提醒。

继电器：像一个“电磁开关”，Arduino 用小电流控制继电器线圈，继电器再 接通水泵的大电流回路，从而安全地控制水泵。泵的负极经过继电器的 COM 与 NO 触点接入回路。

LCD1602：通过 IC1602 转接板把并口屏幕变成 I2C 接口，只占用 SDA、SCL 两根 信号线，实时显示光照值、土壤湿度和“灯/泵”的开关状态。

3运行代码

依赖库：LiquidCrystal_I2C（在 Arduino IDE“库管理器”中 搜索安装）。Wire 为内置库，无需安装。

smart_farm.ino 
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// ===== 引脚定义 =====
#define LIGHT_SENSE_PIN  A1   // 光敏模块 AO（环境光）
#define SOIL_SENSE_PIN   A0   // 土壤湿度 AO（模拟量）
#define SOIL_DO_PIN      D2   // 土壤湿度 DO（数字量，阈值由蓝色电位器决定）
#define LAMP_PIN         D7   // 照明灯（5mm LED，串 220Ω）
#define RELAY_PIN        D8   // 继电器 IN，控制水泵
#define DRY_LED_PIN      13   // 板载 LED D13：土壤干燥时点亮提醒

// ===== 标定值：先用串口观察 ADC，再填入实际的干/湿读数 =====
#define DRY_VALUE  800    // 完全干燥时的 ADC（干 -> 数值高）
#define WET_VALUE  300    // 完全湿润时的 ADC（湿 -> 数值低）

// ===== 阈值与判定 =====
#define LIGHT_THRESHOLD  500  // 光照低于此值 -> 天黑开灯
#define DRY_PCT          30   // 湿度低于 30% -> 过干，启动水泵
#define WET_PCT          70   // 湿度高于 70% -> 过湿

// ===== 方向/触发：不同模块可能相反，按串口读数改 =====
#define DARK_IS_LOW      true  // true：光照越弱 AO 越小
#define DO_ACTIVE_LOW    true  // true：干燥时 DO 输出低电平（常见）
#define RELAY_ACTIVE_LOW true  // true：继电器低电平触发（多数模块）

#define UPDATE_DELAY     2000UL  // 每 2 秒更新一次，想更慢就调大

LiquidCrystal_I2C lcd(0x27, 16, 2);

bool lampOn = false;
bool pumpOn = false;

void setRelay(bool on) {
  bool active = RELAY_ACTIVE_LOW ? LOW : HIGH;   // 继电器触发电平
  digitalWrite(RELAY_PIN, on ? active : !active);
}

void lcdPad(int n) {   // 数字右对齐成 4 位
  if (n < 10)   lcd.print(' ');
  if (n < 100)  lcd.print(' ');
  if (n < 1000) lcd.print(' ');
  lcd.print(n);
}

void lcdPct(int v) {   // 百分比右对齐成 3 位
  if (v < 10)       lcd.print("  ");
  else if (v < 100) lcd.print(' ');
  lcd.print(v);
}

void setup() {
  pinMode(LAMP_PIN, OUTPUT);
  pinMode(RELAY_PIN, OUTPUT);
  pinMode(DRY_LED_PIN, OUTPUT);
  pinMode(SOIL_DO_PIN, INPUT);
  digitalWrite(LAMP_PIN, LOW);
  digitalWrite(DRY_LED_PIN, LOW);
  setRelay(false);            // 初始：灯灭、泵停、干燥灯灭

  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("Smart Farm v1.1");
  lcd.setCursor(0, 1);
  lcd.print("Starting...");
  delay(1500);

  Serial.begin(9600);
  Serial.println("Smart Farm ready");
}

void loop() {
  // ---- 光照 ----
  int light = analogRead(LIGHT_SENSE_PIN);
  bool dark = DARK_IS_LOW ? (light < LIGHT_THRESHOLD)
                          : (light > LIGHT_THRESHOLD);
  lampOn = dark;
  digitalWrite(LAMP_PIN, lampOn ? HIGH : LOW);

  // ---- 土壤湿度 ----
  int soil = analogRead(SOIL_SENSE_PIN);
  int doLevel = digitalRead(SOIL_DO_PIN);
  float voltage = soil * 5.0 / 1023.0;
  int moisture = constrain(map(soil, DRY_VALUE, WET_VALUE, 0, 100), 0, 100);
  bool doDry = DO_ACTIVE_LOW ? (doLevel == LOW) : (doLevel == HIGH);
  const char* doStatus = doDry ? "干燥" : "湿润";
  const char* status = moisture < DRY_PCT ? "过干"
                     : moisture > WET_PCT ? "过湿" : "适中";

  bool dry = (moisture < DRY_PCT);
  pumpOn = dry;
  setRelay(pumpOn);
  digitalWrite(DRY_LED_PIN, dry ? HIGH : LOW);

  // ---- LCD 显示 ----
  lcd.setCursor(0, 0);
  lcd.print("L:"); lcdPad(light);
  lcd.print(" Lamp:"); lcd.print(lampOn ? "ON  " : "OFF ");

  lcd.setCursor(0, 1);
  lcd.print("S:"); lcdPct(moisture); lcd.print("%");
  lcd.print(" Pump:"); lcd.print(pumpOn ? "ON  " : "OFF ");

  // ---- 串口：打印土壤 6 项 + 光照 ----
  Serial.println("===== 土壤湿度 =====");
  Serial.print("1. ADC 原始值 : "); Serial.println(soil);
  Serial.print("2. 输出电压  : "); Serial.print(voltage, 2); Serial.println(" V");
  Serial.print("3. 土壤湿度  : "); Serial.print(moisture); Serial.println(" %");
  Serial.print("4. DO 电平   : "); Serial.println(doLevel);
  Serial.print("5. DO 状态   : "); Serial.println(doStatus);
  Serial.print("6. 综合状态  : "); Serial.println(status);
  Serial.print("光照 ADC="); Serial.print(light);
  Serial.print(" 灯="); Serial.print(lampOn ? "开" : "关");
  Serial.print(" 泵="); Serial.println(pumpOn ? "开" : "关");
  Serial.println("------------------------");

  delay(UPDATE_DELAY);
}
第一次使用先标定：上传后打开串口监视器（波特率 9600），把土壤探头分别插入完全干燥的土和浇透水的土，记下 两端的 ADC 值，填进 DRY_VALUE（干）和 WET_VALUE（湿）。若 DO 的“干燥/湿润”判断反了，把 DO_ACTIVE_LOW 取反；继电器或灯光方向反了，分别改 RELAY_ACTIVE_LOW / DARK_IS_LOW。 若屏幕无任何显示，把代码里 lcd(0x27, 16, 2) 的地址改成 0x3F 再试（这是 LCD1602 I2C 常见的两种地址）。

5NodeMCU 网页服务器程序

这是烧到 NodeMCU (ESP8266) 上的独立程序：开 WiFi 热点 SmartFarm、提供网页显示传感器数据并调节照明灯亮度。 用 Arduino IDE 选 NodeMCU 1.0 (ESP-12E Module) 上传 （需先在“开发板管理器”安装 esp8266）。

SmartFarmWeb.ino 
/*
  SmartFarm WiFi 网页服务器 —— NodeMCU (ESP8266)
  功能：开 WiFi 热点 + 网页；显示传感器数据、调节照明灯亮度。
  与 Arduino UNO 通过串口(UART0 = GPIO1/3)通信：
    Arduino -> NodeMCU : SENSOR,光,暗,灯,土,泵,距离
    NodeMCU -> Arduino : BRIGHT,亮度值(0-255)
  烧录：Arduino IDE -> 开发板选 "NodeMCU 1.0 (ESP-12E Module)" -> 用 USB 上传
*/
#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>

// ===== 热点配置 =====
const char* AP_SSID = "SmartFarm";
const char* AP_PASS = "12345678";   // 至少 8 位

// ===== 传感器状态（Arduino 每 2 秒发来）=====
int sLight = 0;      // 光敏 DO 电平
int sDark = 0;       // 黑暗判定(1=暗)
int sLamp = 0;       // 照明灯开关(1=开)
int sSoil = 0;       // 土壤湿度 %
int sPump = 0;       // 水泵开关(1=开)
float sDist = -1;    // 距离 cm
int sBright = 100;   // 灯亮度 0-255

ESP8266WebServer server(80);

// 解析 Arduino 发来的 "SENSOR,光,暗,灯,土,泵,距离"
void parseLine(String line) {
  if (!line.startsWith("SENSOR,")) return;
  String b = line.substring(7);
  int v[6], idx = 0, start = 0;
  for (int i = 0; i <= b.length(); i++) {
    if (i == b.length() || b.charAt(i) == ',') {
      if (idx < 6) v[idx++] = b.substring(start, i).toInt();
      start = i + 1;
    }
  }
  if (idx >= 6) {
    sLight = v[0]; sDark = v[1]; sLamp = v[2];
    sSoil = v[3]; sPump = v[4]; sDist = v[5];
  }
}

void handleRoot() { server.send(200, "text/html", ROOT_HTML); }

// 返回传感器 JSON（网页轮询用）
void handleData() {
  String j = "{\"light\":" + String(sLight)
    + ",\"dark\":" + String(sDark)
    + ",\"lamp\":" + String(sLamp)
    + ",\"soil\":" + String(sSoil)
    + ",\"pump\":" + String(sPump)
    + ",\"dist\":" + String(sDist, 0)
    + ",\"bright\":" + String(sBright) + "}";
  server.send(200, "application/json", j);
}

// 网页亮度滑块：收到值后发命令给 Arduino
void handleBright() {
  if (server.hasArg("val")) {
    int v = server.arg("val").toInt();
    if (v < 0) v = 0;
    if (v > 255) v = 255;
    sBright = v;
    Serial.print("BRIGHT,"); Serial.println(v);   // 发给 Arduino
  }
  server.send(200, "text/plain", String(sBright));
}

void setup() {
  Serial.begin(9600);             // 与 Arduino 通信（UART0=GPIO1/3）
  WiFi.softAP(AP_SSID, AP_PASS);  // 开热点
  server.on("/", handleRoot);
  server.on("/data", handleData);
  server.on("/bright", handleBright);
  server.begin();
}

void loop() {
  // 读 Arduino 发来的传感器数据
  while (Serial.available()) {
    String line = Serial.readStringUntil('\n');
    line.trim();
    parseLine(line);
  }
  server.handleClient();
}
